// ---- primitive factories ----
type Perm = { tag: string; operator: string; value: unknown };
const perm = (tag: string, operator: string, value: unknown): Perm => ({ tag, operator, value });

const io = (
  tag: string,
  signal_type: string,
  io_address: string,
  description: string,
  source: "wired" | "network_telegram" = "wired",
) => ({ tag, signal_type, io_address, description, source, tier: source === "wired" ? "wired" : "fb_instance" });

const cmEntry = (tag: string, description: string, state: string) => ({ tag, description, state });

const staticState = (control_modules: ReturnType<typeof cmEntry>[], notes: string | null = null) => ({
  control_modules,
  notes,
});

const crit = {
  tagEq: (tag: string, value: unknown) => ({ kind: "tag_equals", tag, value }),
  expr: (text: string, referenced_tags: string[]) => ({ kind: "expression", text, referenced_tags }),
  ack: (prompt: string) => ({ kind: "manual_ack", prompt }),
};

const act = {
  prose: (action_id: string, text: string, referenced_tags: string[]) => ({
    kind: "manual_prose",
    action_id,
    text,
    referenced_tags,
    prose: text,
  }),
  assign: (action_id: string, target_tag: string, litValue: unknown, prose: string) => ({
    kind: "assign",
    action_id,
    target_tag,
    source:
      typeof litValue === "string"
        ? { kind: "literal", value: litValue, value_type: "string" }
        : typeof litValue === "boolean"
          ? { kind: "literal", value: litValue, value_type: "boolean" }
          : { kind: "literal", value: litValue, value_type: "number" },
    prose,
  }),
};

const step = (
  n: number,
  name: string,
  action: string,
  completion_criteria_text: string,
  completion_criteria: unknown[] = [],
  actions: unknown[] = [],
) => ({
  step_id: `s${n}`,
  branch_id: "main",
  name,
  actions,
  monitors: [],
  transitions: [],
  step: n,
  action,
  completion_criteria,
  completion_criteria_text,
  on_fail: undefined,
});

const seqState = (permissives: Perm[], steps: unknown[], notes: string | null = null) => ({
  permissives,
  steps,
  branches: [{ branch_id: "main", name: "Main", kind: "main", fork_step_id: "s1" }],
  state_monitors: [],
  sequence_model_version: 2,
  notes,
});

// Standard PackML-aligned EM state set (safe state = "aborted") + transitions.
const stdStates = () => [
  { state_id: "idle", name: "Idle", kind: "static", allowed_modes: [], is_safe_state: false },
  { state_id: "execute", name: "Execute", kind: "sequential", allowed_modes: [], is_safe_state: false },
  { state_id: "aborted", name: "Aborted", kind: "static", allowed_modes: [], is_safe_state: true },
];
const stdTransitions = (startTag = "TOP.CTRL.Automatic_Run") => [
  { transition_id: "idle_to_execute", from_state_id: "idle", to_state_id: "execute", trigger: { kind: "command", expr: perm(startTag, "=", true) }, guard: [] },
  { transition_id: "execute_to_idle", from_state_id: "execute", to_state_id: "idle", trigger: { kind: "completion" }, guard: [] },
  { transition_id: "execute_to_aborted", from_state_id: "execute", to_state_id: "aborted", trigger: { kind: "command", expr: perm("TOP.CTRL.Fault[0].MajorFault", "=", true) }, guard: [] },
  { transition_id: "idle_to_aborted", from_state_id: "idle", to_state_id: "aborted", trigger: { kind: "command", expr: perm("TOP.CTRL.Fault[0].MajorFault", "=", true) }, guard: [] },
  { transition_id: "aborted_to_idle", from_state_id: "aborted", to_state_id: "idle", trigger: { kind: "command", expr: perm("TOP.CTRL.Fault_Reset_Request", "=", true) }, guard: [] },
];

// ---------- Equipment Modules ----------
// Each entry: { id, hier (EquipmentModuleV2), contract (EquipmentModuleContract) }
function makeEm(opts: {
  name: string;
  description: string;
  cms: unknown[];
  states?: unknown[];
  transitions?: unknown[];
  staticStates: Record<string, unknown>;
  seqStates: Record<string, unknown>;
  commandBehavior?: Record<string, unknown>;
}) {
  const id = uid();
  const states = opts.states ?? stdStates();
  const transitions = opts.transitions ?? stdTransitions();
  return {
    id,
    hier: { equipment_module_id: id, equipment_module_name: opts.name, description: opts.description, control_modules: opts.cms },
    contract: {
      equipment_module_id: id,
      unit_id: UNIT_ID,
      states,
      transitions,
      static_states: opts.staticStates,
      sequential_states: opts.seqStates,
      ...(opts.commandBehavior ? { command_behavior: opts.commandBehavior } : {}),
    },
  };
}

describe("Sun Metals Z20 KUKA_ROBOT — Spec Contract V2 build", () => {
  it("validates against SpecContractV2Schema and EM invariants, then writes JSON", () => {
    // EM state-machine invariants (one safe state, unique ids, valid transitions)
    const emIssues: string[] = [];
    for (const e of EMS) emIssues.push(...validateEmStateMachine(e.contract as never));
    expect(emIssues, `EM state-machine issues:\n${emIssues.join("\n")}`).toEqual([]);

    const parsed = SpecContractV2Schema.safeParse(contract);
    if (!parsed.success) {
      // Surface the first handful of issues clearly.
      const msg = parsed.error.issues
        .slice(0, 25)
        .map((i) => `${i.path.join(".")}: ${i.message}`)
        .join("\n");
      throw new Error(`Contract failed schema validation:\n${msg}`);
    }

    const out = "C:/dev/pac-forge/test-data/Sun_Metals_Z20_KUKA_ROBOT.spec-v2.json";
    writeFileSync(out, JSON.stringify(parsed.data, null, 2), "utf8");
    expect(parsed.success).toBe(true);
  });
});
